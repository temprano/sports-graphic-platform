**Status**: Design Phase

## 🎯 Complete Order Creation Flow

```
Step 1: Brand Selection
    ↓ User selects "Cinematic Dark"
Step 2: Player Selection  
    ↓ User adds 5 players from roster
Step 3: Photo Upload ← THIS DESIGN
    ↓ Admin uploads 3-5 poses per player
Step 4: Review & Confirmation
    ↓ Admin reviews order summary
Step 5: Payment (Deposit)
    ↓ Team member pays deposit via Stripe
Step 6: Order Submitted
    ↓ Triggers BullMQ processing
    ├─→ process-photos.js (organize uploaded photos)
    ├─→ render-video.js (create motion graphics)
    ├─→ render-print.js (create print templates)
    └─→ package-order.js (bundle deliverables)
Step 7: Proof Review
    ↓ Admin reviews generated proofs
Step 8: Final Payment & Fulfillment
    ↓ Full payment processed
    ↓ Fulfillment to Prodigi/Printful
```

## 📋 Photo Upload → Pipeline Connection

### 1. Upload Photos (Web App)

**Component**: `PhotoUploadFlow`  
**User**: Team Admin  
**Action**: Uploads 15 photos (5 players × 3 poses)

**Data Submitted**:
```javascript

  players: [
    {
      playerId: "player-1",
      name: "Jordan Smith",

        "left-angle": { file: File, confidence: 0.88 },
        "right-angle": { file: File, confidence: 0.91 }
      }
    }
    // ... 4 more players
  ]
}
```

### 2. Store Photos (Backend API)

**Route**: `POST /api/orders`

**Process**:
```javascript
async function createOrder(req, res) {
  const { orderId, players } = req.body;
  
  // 1. Validate all photos are present
  if (!allPhotosPresent(players)) {
    return res.status(400).json({ error: 'Missing photos' });
  }
  
  // 2. Upload to Appwrite storage
  const photoUrls = await Promise.all(
    players.flatMap(player =>
      Object.entries(player.poses).map(([poseId, photoData]) =>
        appwrite.storage.createFile(
          bucketId,
          `${orderId}/raw-photos/${player.playerId}/${poseId}.jpg`,
          photoData.file
        )
      )
    )
  );
  
  // 3. Create order document in Appwrite
  const order = await appwrite.database.createDocument('orders', {
    id: orderId,
    team_id,
    status: 'IN_PRODUCTION',
    photos: photoUrls,
    deliverables: [
      { type: 'team-banner', status: 'pending' },
      { type: 'player-banners', status: 'pending' },
      { type: 'player-posters', status: 'pending' }
    ]
  });
  
  // 4. Queue processing job
  await queue.add('process-photos', {
    orderId,
    players,
    photoUrls
  });
  
  res.json({ orderId, status: 'processing' });
}
```

### 3. Process Photos (BullMQ Job)

**Job**: `src/queue/jobs/process-photos.js`

**Steps**:
```javascript
async function processPhotos(job) {
  const { orderId, players, photoUrls } = job.data;
  
  // 1. Create local processing folder
  const processingDir = `~/src/renderPhotos/${orderId}`;
  mkdir(processingDir);
  
  // 2. Download photos from Appwrite
  for (const player of players) {
    const playerDir = `${processingDir}/${player.playerId}`;
    mkdir(playerDir);
    
    for (const [poseId, photoUrl] of Object.entries(photoUrls[player.playerId])) {
      // Download to local disk
      downloadFile(photoUrl, `${playerDir}/${poseId}.jpg`);
    }
    
    // Write metadata
    writeJSON(`${playerDir}/metadata.json`, {
      playerId: player.playerId,
      name: player.name,
      poses: {
        "front-facing": { source: "front-facing.jpg", validAt: new Date() },
        "left-angle": { source: "left-angle.jpg", validAt: new Date() },
        "right-angle": { source: "right-angle.jpg", validAt: new Date() }
      }
    });
  }
  
  // 3. Run optional quality check
  // (Re-validate with posenet server-side if strict validation needed)
  
  // 4. Mark photos as ready
  // 5. Queue rendering jobs
  for (const player of players) {
    await queue.add('render-video', {
      orderId,
      playerId: player.playerId,
      photosPath: `${processingDir}/${player.playerId}`,
      brand: order.brand_folder
    });
    
    await queue.add('render-print', {
      orderId,
      playerId: player.playerId,
      photosPath: `${processingDir}/${player.playerId}`,
      formats: ['poster-16x20', 'banner-2x6', 'card-4x6']
    });
  }
  
  return { status: 'photos-processed', orderId };
}
```

## 🔐 Security & Permissions

### Upload Authorization

```typescript
// Middleware: /api/orders (POST)
async function requireTeamAdmin(req, res, next) {
  const user = await getAuthUser(req);
  const team = await appwrite.database.getDocument('teams', user.team_id);
  
  if (user.role !== 'admin') {
    return res.status(403).json({ error: 'Only team admins can upload photos' });
  }
  
  next();
}
```

### Storage Bucket Rules

```
Bucket: "team-photos" (private)
├── Read:
│   - Allow if user.team_id == bucket.team_id (team members)
│   - Allow if job.type == "render-video" (pipeline)
│   - Allow if job.type == "render-print" (pipeline)
│
└── Write:
    - Allow if user.role == "admin" AND user.team_id == bucket.team_id
    - Deny all others
```

### Data Retention

```javascript
// Auto-delete after 90 days
// Except: Finals stored indefinitely (for fulfillment history)
storageCleanupJob.schedule('0 */6 * * *', async () => {
  const ninetyDaysAgo = new Date(Date.now() - 90 * 24 * 60 * 60 * 1000);
  
  for (const file of allFiles) {
    if (file.type === 'raw-photo' && file.uploadedAt < ninetyDaysAgo) {
      appwrite.storage.deleteFile(file.id);
    }
  }
});
```
